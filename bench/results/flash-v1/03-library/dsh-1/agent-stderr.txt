```js
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

export function isDate(value) {
  if (typeof value !== 'string' || !DATE_RE.test(value)) return false;
  const [y, m, d] = value.split('-').map(Number);
  const date = new Date(Date.UTC(y, m - 1, d));
  return (
    date.getUTCFullYear() === y &&
    date.getUTCMonth() === m - 1 &&
    date.getUTCDate() === d
  );
}

export function assertDate(value, name = 'date') {
  if (!isDate(value)) {
    throw new Error(`${name} must be a valid date in YYYY-MM-DD format`);
  }
  return value;
}

export function today() {
  const now = new Date();
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, '0');
  const d = String(now.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

function toUTC(iso) {
  const [y, m, d] = iso.split('-').map(Number);
```

```js
    throw new Error('data file path must be a non-empty string');

  return dataFile;

export function emptyData() {
  return {
    books: [],
    members: [],
    loans: [],
    counters: { book: 0, member: 0, loan: 0 },
  };
}

export function readData() {
  if (!existsSync(dataFile)) {
    const data = emptyData();
    writeData(data);
    return data;
  }
  const raw = readFileSync(dataFile, 'utf8');
  let parsed;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new Error(`could not parse data file: ${dataFile}`);
  }
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`invalid data file: ${dataFile}`);
  }
  const data = emptyData();
  for (const key of ['books', 'members', 'loans']) {
    if (Array.isArray(parsed[key])) data[key] = parsed[key];
  }
  if (parsed.counters && typeof parsed.counters === 'object') {
    for (const key of ['book', 'member', 'loan']) {
      const value = Number(parsed.counters[key]);
      if (Number.isInteger(value) && value >= 0) data.counters[key] = value;
    }
  }
  return data;
}

export function writeData(data) {
  mkdirSync(dirname(dataFile), { recursive: true });
  writeFileSync(dataFile, JSON.stringify(data, null, 2) + '\n');
}

export function nextId(data, type, items) {
  const maxExisting = items.reduce((max, item) => (Number.isInteger(item.id) && item.id > max ? item.id : max), 0);
  const next = Math.max(data.counters[type] || 0, maxExisting) + 1;
  data.counters[type] = next;
  return next;
}
```

```js
import { addDays, assertDate, daysBetween, todayISO } from './dates.js';
import { nextId, readData, writeData } from './storage.js';

export const MAX_ACTIVE_LOANS = 3;
export const FINE_PER_DAY = 0.5;
export const MAX_FINE = 20;

function toId(value, name) {
  const id = Number(value);
  if (!Number.isInteger(id) || id < 1) {
    throw new Error(`${name} must be a positive integer`);
  }
  return id;
}

function resolveToday(value) {
  const day = value === undefined ? todayISO() : value;
  return assertDate(day, 'today');
}

function activeLoansFor(data, memberId) {
  return data.loans.filter((loan) => loan.memberId === memberId && loan.returned === null);
}

export function checkout({ bookId, memberId, days = 14, today } = {}) {
  const out = resolveToday(today);
  const bookKey = toId(bookId, 'bookId');
  const memberKey = toId(memberId, 'memberId');
  const loanDays = Number(days);
  if (!Number.isInteger(loanDays) || loanDays < 1) {
    throw new Error('days must be a positive integer');
  }
  const data = readData();
  const book = data.books.find((b) => b.id === bookKey);
  if (!book) throw new Error(`book ${bookKey} not found`);
  const member = data.members.find((m) => m.id === memberKey);
  if (!member) throw new Error(`member ${memberKey} not found`);
  if (book.available < 1) throw new Error(`book ${bookKey} has no available copies`);
  if (activeLoansFor(data, memberKey).length >= MAX_ACTIVE_LOANS) {
    throw new Error(`member ${memberKey} already has ${MAX_ACTIVE_LOANS} active loans`);
  }
  const id = nextId(data, 'loan', data.loans);
  const loan = { id, bookId: bookKey, memberId: memberKey, out, due: addDays(out, loanDays), returned: null, fine: 0 };
  data.loans.push(loan);
  book.available -= 1;
  writeData(data);
  return { ...loan };
}

export function returnLoan({ loanId, today } = {}) {
  const returnedOn = resolveToday(today);
  const key = toId(loanId, 'loanId');
  const data = readData();
  const loan = data.loans.find((l) => l.id === key);
  if (!loan) throw new Error(`loan ${key} not found`);
  if (loan.returned !== null) throw new Error(`loan ${key} has already been returned`);
  const lateDays = daysBetween(loan.due, returnedOn);
  loan.returned = returnedOn;
  loan.fine = lateDays > 0 ? Math.min(lateDays * FINE_PER_DAY, MAX_FINE) : 0;
  const book = data.books.find((b) => b.id === loan.bookId);
  if (book && book.available < book.copies) book.available += 1;
  writeData(data);
  return { ...loan };
}

export function listLoans({ memberId, overdue, today } = {}) {
  const day = resolveToday(today);
  const data = readData();
  let loans = data.loans;
  if (memberId !== undefined && memberId !== null) {
    const key = toId(memberId, 'memberId');
    loans = loans.filter((loan) => loan.memberId === key);
  }
  if (overdue) {
    loans = loans.filter((loan) => loan.returned === null && daysBetween(loan.due, day) > 0);
  }
  return loans
    .slice()
    .sort((a, b) => a.id - b.id)
    .map((loan) => ({ ...loan }));
}

export function report({ today } = {}) {
  const day = resolveToday(today);
  const data = readData();
  const books = data.books.length;
  const copies = data.books.reduce((sum, book) => sum + book.copies, 0);
  const available = data.books.reduce((sum, book) => sum + book.available, 0);
  const members = data.members.length;
  const activeLoans = data.loans.filter((loan) => loan.returned === null).length;
  const overdue = data.loans.filter((loan) => loan.returned === null && daysBetween(loan.due, day) > 0).length;
  const finesOwed = data.loans
    .filter((loan) => loan.returned !== null)
    .reduce((sum, loan) => sum + (Number(loan.fine) || 0), 0);
  return { books, copies, available, members, activeLoans, overdue, finesOwed };
}
```
