```js
import fs from 'node:fs';
import path from 'node:path';

let dataPath = path.resolve(process.cwd(), 'data', 'library.json');

export function setDataPath(p) {
  dataPath = path.resolve(p);
}
export function getDataPath() { return dataPath; }

function emptyState() {
  return {
    books: [],
    members: [],
    loans: [],
    sequences: { books: 0, members: 0, loans: 0 },
  };
}

export function load() {
  if (!fs.existsSync(dataPath)) return emptyState();
  const text = fs.readFileSync(dataPath, 'utf8');
  if (text.trim() === '') return emptyState();
  const state = JSON.parse(text);
  if (!state.sequences) state.sequences = { books: 0, members: 0, loans: 0 };
  for (const key of ['books','members','loans']) if (!Array.isArray(state[key])) state[key] = [];
  return state;
}

export function save(state) {
  fs.mkdirSync(path.dirname(dataPath), { recursive: true });
  fs.writeFileSync(dataPath, JSON.stringify(state, null, 2) + '\n');
}

export function ensureStore() {
  if (!fs.existsSync(dataPath)) save(emptyState());
}

function nextId(state, key) {
  const plural = key; // 'books'
  const current = state.sequences[plural] ?? 0;
  const max = state[plural].reduce((m, item) => Math.max(m, item.id ?? 0), 0);
  const id = Math.max(current, max) + 1;
  state.sequences[plural] = id;
  return id;
}

export { nextId };
```

```js
import { load, save, nextId } from './store.js';
import { LibraryError } from './errors.js';
import { todayISO, addDays, diffDays, isValidDate } from './dates.js';

const MAX_ACTIVE = 3;
const FINE_PER_DAY = 0.5;
const FINE_CAP = 20;

function resolveToday(today) {
  if (today === undefined || today === null) return todayISO();
  if (!isValidDate(today)) throw new LibraryError('today must be a valid YYYY-MM-DD date');
  return today;
}

export function checkout({ bookId, memberId, days = 14, today } = {}) {
  if (!Number.isInteger(bookId) || bookId < 1) throw new LibraryError('bookId is required');
  if (!Number.isInteger(memberId) || memberId < 1) throw new LibraryError('memberId is required');
  const d = days === undefined ? 14 : days;
  if (!Number.isInteger(d) || d < 1) throw new LibraryError('days must be a positive integer');
  const out = resolveToday(today);
  const state = load();
  const book = state.books.find(b => b.id === bookId);
  if (!book) throw new LibraryError(`book ${bookId} not found`);
  const member = state.members.find(m => m.id === memberId);
  if (!member) throw new LibraryError(`member ${memberId} not found`);
  if (book.available < 1) throw new LibraryError(`book ${bookId} has no available copies`);
  const active = state.loans.filter(l => l.memberId === memberId && l.returned === null);
  if (active.length >= MAX_ACTIVE) throw new LibraryError(`member ${memberId} already has ${MAX_ACTIVE} active loans`);
  const loan = { id: nextId(state,'loans'), bookId, memberId, out, due: addDays(out, d), returned: null, fine: 0 };
  state.loans.push(loan);
  book.available -= 1;
  save(state);
  return loan;
}

export function returnLoan({ loanId, today } = {}) {
  if (!Number.isInteger(loanId) || loanId < 1) throw new LibraryError('loanId is required');
  const returned = resolveToday(today);
  const state = load();
  const loan = state.loans.find(l => l.id === loanId);
  if (!loan) throw new LibraryError(`loan ${loanId} not found`);
  if (loan.returned !== null) throw new LibraryError(`loan ${loanId} has already been returned`);
  loan.returned = returned;
  const late = diffDays(returned, loan.due);
  loan.fine = late > 0 ? Math.min(late * FINE_PER_DAY, FINE_CAP) : 0;
  const book = state.books.find(b => b.id === loan.bookId);
  if (book) book.available = Math.min(book.available + 1, book.copies);
  save(state);
  return loan;
}

export function listLoans({ memberId, overdue, today } = {}) {
  const t = resolveToday(today);
  let loans = load().loans;
  if (memberId !== undefined) {
    if (!Number.isInteger(memberId)) throw new LibraryError('memberId must be an integer');
    loans = loans.filter(l => l.memberId === memberId);
  }
  if (overdue) loans = loans.filter(l => l.returned === null && l.due < t);
  return loans;
}

export function report({ today } = {}) {
  const t = resolveToday(today);
  const state = load();
  const active = state.loans.filter(l => l.returned === null);
  const overdue = active.filter(l => l.due < t).length;
  const finesOwed = state.loans.filter(l => l.returned !== null).reduce((s,l)=>s+(l.fine||0),0);
  return {
    books: state.books.length,
    copies: state.books.reduce((s,b)=>s+b.copies,0),
    available: state.books.reduce((s,b)=>s+b.available,0),
    members: state.members.length,
    activeLoans: active.length,
    overdue,
    finesOwed: Math.round(finesOwed*100)/100,
  };
}
```

```js
function parse(argv) {
  const command = argv[0];
  if (!command || command.startsWith('--')) throw new UsageError(command ? `unknown command: ${command}` : 'missing command');
  const spec = COMMANDS[command];
  if (!spec) throw new UsageError(`unknown command: ${command}`);
  const opts = {};
  for (let i = 1; i < argv.length; i++) {
    const arg = argv[i];
    if (!arg.startsWith('--')) throw new UsageError(`unexpected argument: ${arg}`);
    const name = arg.slice(2);
    const type = spec[name];
    if (!type) throw new UsageError(`unknown option: ${arg}`);
    if (type === 'flag') { opts[name] = true; continue; }
    // value option
    const next = argv[i+1];
    if (next === undefined || (next.startsWith('--') && spec[next.slice(2)] !== undefined)) {
      throw new UsageError(`missing value for option: ${arg}`);
    }
    i++;
    opts[name] = type === 'int' ? Number(next) : next;
  }
  return { command, opts };
}
```
