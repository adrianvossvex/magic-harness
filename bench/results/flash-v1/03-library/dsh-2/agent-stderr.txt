```js
import fs from 'node:fs';
import path from 'node:path';

export const DEFAULT_DATA_FILE = 'data/library.json';

export function resolveDataPath(explicit) {
  if (explicit) return path.resolve(explicit);
  if (process.env.LIBRARY_DATA) return path.resolve(process.env.LIBRARY_DATA);
  return path.resolve(DEFAULT_DATA_FILE);
}

export function emptyDb() {
  return { books: [], members: [], loans: [] };
}

function normalizeList(value) {
  return Array.isArray(value) ? value : [];
}

export function syncAvailability(db) {
  for (const book of db.books) {
    const active = db.loans.filter(l => l.bookId === book.id && l.returned === null).length;
    book.available = Math.max(0, book.copies - active);
  }
  return db;
}

export function loadDb(explicit) {
  const file = resolveDataPath(explicit);
  let db;
  if (!fs.existsSync(file)) {
    db = emptyDb();
  } else {
    const raw = fs.readFileSync(file, 'utf8');
    if (raw.trim() === '') db = emptyDb();
    else {
      try { db = JSON.parse(raw); }
      catch { throw new Error(`Invalid data file: ${file}`); }
    }
  }
  db.books = normalizeList(db.books);
  db.members = normalizeList(db.members);
  db.loans = normalizeList(db.loans);
  return syncAvailability(db);
}

export function saveDb(db, explicit) {
  const file = resolveDataPath(explicit);
  syncAvailability(db);
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.writeFileSync(file, JSON.stringify(db, null, 2) + '\n', 'utf8');
  return db;
}

export function nextId(items) {
  let max = 0;
  for (const item of items) if (Number.isInteger(item?.id) && item.id > max) max = item.id;
  return max + 1;
}
```

```js
export function todayISO(now = new Date()) {
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, '0');
  const d = String(now.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

export function isValidDateString(value) {
  if (typeof value !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(value)) return false;
  const [y, m, d] = value.split('-').map(Number);
  const dt = new Date(Date.UTC(y, m - 1, d));
  return dt.getUTCFullYear() === y && dt.getUTCMonth() === m - 1 && dt.getUTCDate() === d;
}

export function parseDate(iso) {
  const [y, m, d] = iso.split('-').map(Number);
  return Date.UTC(y, m - 1, d);
}

export function addDays(iso, days) {
  const dt = new Date(parseDate(iso));
  dt.setUTCDate(dt.getUTCDate() + days);
  return dt.toISOString().slice(0, 10);
}

export function diffDays(a, b) {
  return Math.round((parseDate(a) - parseDate(b)) / 86400000);
}
```

```js
const MAX_ACTIVE_LOANS = 3;
const FINE_PER_DAY = 0.5;
const MAX_FINE = 20;

function resolveToday(today) {
  if (today === undefined || today === null) return todayISO();
  if (!isValidDateString(today)) throw new Error(`invalid date: ${today}`);
  return today;
}

function toId(value, name) {
  const n = Number(value);
  if (!Number.isInteger(n) || n < 1) throw new Error(`${name} must be a positive integer`);
  return n;
}

export function checkout({ bookId, memberId, days = 14, today } = {}, dataPath) {
  const db = loadDb(dataPath);
  const bid = toId(bookId, 'bookId');
  const mid = toId(memberId, 'memberId');
  const book = db.books.find(b => b.id === bid);
  if (!book) throw new Error(`book ${bid} not found`);
  const member = db.members.find(m => m.id === mid);
  if (!member) throw new Error(`member ${mid} not found`);
  const activeForBook = db.loans.filter(l => l.bookId === bid && l.returned === null).length;
  if (activeForBook >= book.copies) throw new Error(`no available copy of book ${bid}`);
  const activeForMember = db.loans.filter(l => l.memberId === mid && l.returned === null).length;
  if (activeForMember >= MAX_ACTIVE_LOANS) throw new Error(`member ${mid} already has ${MAX_ACTIVE_LOANS} active loans`);
  const outDate = resolveToday(today);
  const d = Number(days);
  if (!Number.isInteger(d) || d < 1) throw new Error('days must be a positive integer');
  const due = addDays(outDate, d);
  const loan = { id: nextId(db.loans), bookId: bid, memberId: mid, out: outDate, due, returned: null, fine: 0 };
  db.loans.push(loan);
  saveDb(db, dataPath);
  return loan;
}

export function returnLoan({ loanId, today } = {}, dataPath) {
  const db = loadDb(dataPath);
  const id = toId(loanId, 'loanId');
  const loan = db.loans.find(l => l.id === id);
  if (!loan) throw new Error(`loan ${id} not found`);
  if (loan.returned !== null) throw new Error(`loan ${id} has already been returned`);
  const returned = resolveToday(today);
  const late = diffDays(returned, loan.due);
  const fine = late > 0 ? Math.min(late * FINE_PER_DAY, MAX_FINE) : 0;
  loan.returned = returned;
  loan.fine = Number(fine.toFixed(2));
  saveDb(db, dataPath);
  return loan;
}

export function listLoans({ memberId, overdue, today } = {}, dataPath) {
  const db = loadDb(dataPath);
  const ref = resolveToday(today);
  let loans = db.loans.slice();
  if (memberId !== undefined && memberId !== null) {
    const mid = toId(memberId, 'memberId');
    loans = loans.filter(l => l.memberId === mid);
  }
  if (overdue) loans = loans.filter(l => l.returned === null && l.due < ref);
  return loans;
}

export function report({ today } = {}, dataPath) {
  const db = loadDb(dataPath);
  const ref = resolveToday(today);
  const activeLoans = db.loans.filter(l => l.returned === null);
  const overdue = activeLoans.filter(l => l.due < ref);
  const finesOwed = db.loans.filter(l => l.returned !== null).reduce((s, l) => s + (Number(l.fine) || 0), 0);
  return {
    books: db.books.length,
    copies: db.books.reduce((s, b) => s + b.copies, 0),
    available: db.books.reduce((s, b) => s + (b.available ?? 0), 0),
    members: db.members.length,
    activeLoans: activeLoans.length,
    overdue: overdue.length,
    finesOwed: Number(finesOwed.toFixed(2)),
  };
}
```
