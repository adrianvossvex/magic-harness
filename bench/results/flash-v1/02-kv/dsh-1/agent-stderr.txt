export function loadStore(filePath) {
  let raw;
  try {
    raw = readFileSync(filePath, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') return {};
    throw err;
  }
  if (raw.trim() === '') return {};
  let data;
  try {
    data = JSON.parse(raw);
  } catch {
    throw new Error(`invalid store file: ${filePath}`);
  }
  if (data === null || typeof data !== 'object' || Array.isArray(data)) {
    throw new Error(`invalid store file: ${filePath}`);
  }
  return data;
}

import { readFileSync, writeFileSync, renameSync, mkdirSync, rmSync } from 'node:fs';
import { dirname, basename, join } from 'node:path';
import { randomBytes } from 'node:crypto';

export function loadStore(filePath) {
  let raw;
  try {
    raw = readFileSync(filePath, 'utf8');
  } catch (err) {
    if (err && err.code === 'ENOENT') return createStore();
    throw err;
  }
  if (raw.trim() === '') return createStore();
  let parsed;
  try { parsed = JSON.parse(raw); }
  catch { throw new Error(`invalid store file: ${filePath}`); }
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`invalid store file: ${filePath}`);
  }
  return Object.assign(createStore(), parsed);
}

function createStore() { return Object.create(null); }

export function saveStore(filePath, store) {
  const dir = dirname(filePath);
  mkdirSync(dir, { recursive: true });
  const tmp = join(dir, `.${basename(filePath)}.${process.pid}.${randomBytes(6).toString('hex')}.tmp`);
  try {
    writeFileSync(tmp, JSON.stringify(store, null, 2) + '\n', 'utf8');
    renameSync(tmp, filePath);
  } catch (err) {
    try { rmSync(tmp, { force: true }); } catch {}
    throw err;
  }
}

export function isExpired(entry, now = Date.now()) {
  if (entry === null || typeof entry !== 'object') return false;
  const exp = entry.expiresAt;
  return typeof exp === 'number' && Number.isFinite(exp) && exp <= now;
}
