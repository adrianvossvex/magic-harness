Options:

function usageError() {
  process.stderr.write(USAGE);
  return 2;
}

function parseArgs(argv) {
  const rest = [];
  let file = null;
  let ttl = null;
  for (let i = 0; i < argv.length; i += 1) {
    const arg = argv[i];
    if (arg === '--file') {
      if (i + 1 >= argv.length) return { error: 'missing value for --file' };
      file = argv[i + 1];
      i += 1;
    } else if (arg === '--ttl') {
      if (i + 1 >= argv.length) return { error: 'missing value for --ttl' };
      ttl = argv[i + 1];
      i += 1;
    } else {
      rest.push(arg);
    }
  }
  return { rest, file, ttl };
}

function resolveFile(file) {
  return file ?? '.kv.json';
}

function fail(message) {
  process.stderr.write(`${message}\n`);
  return 1;
}

function cmdSet(file, args, ttlArg) {
  if (args.length < 1) return usageError();
  const [key, ...valueParts] = args;
  const value = valueParts.join(' ');
  if (!isValidKey(key)) return fail(`invalid key: ${key}`);

  let expiresAt = null;
  if (ttlArg !== null) {
    const seconds = Number(ttlArg);
    if (!Number.isFinite(seconds) || seconds < 0) {
      return fail(`invalid ttl: ${ttlArg}`);
    }
    expiresAt = Date.now() + Math.round(seconds * 1000);
  }

  const data = loadStore(file);
  purgeExpired(data);
  data[key] = { value, expiresAt };
  saveStore(file, data);
  return 0;
}

function cmdGet(file, args) {
  if (args.length !== 1) return usageError();
  const key = args[0];
  if (!isValidKey(key)) return fail(`invalid key: ${key}`);

  const data = loadStore(file);
  const changed = purgeExpired(data);
  const entry = data[key];
  if (!entry) {
    if (changed) saveStore(file, data);
    return fail(`not found: ${key}`);
  }
  if (changed) saveStore(file, data);
  process.stdout.write(`${entry.value}\n`);
  return 0;
}

function cmdDel(file, args) {
  if (args.length !== 1) return usageError();
  const key = args[0];
  if (!isValidKey(key)) return fail(`invalid key: ${key}`);

  const data = loadStore(file);
  const changed = purgeExpired(data);
  if (!data[key]) {
    if (changed) saveStore(file, data);
    return fail(`not found: ${key}`);
  }
  delete data[key];
  saveStore(file, data);
  process.stdout.write('deleted\n');

}

function cmdList(file, args) {
  if (args.length !== 0) return usageError();
  const data = loadStore(file);
  const changed = purgeExpired(data);
  if (changed) saveStore(file, data);
  for (const key of Object.keys(data).sort()) {

  }
  return 0;
}

  if (parsed.error) {
    process.stderr.write(`${parsed.error}\n`);
    return usageError();

import fs from 'node:fs';
import path from 'node:path';
import { randomBytes } from 'node:crypto';

export const KEY_PATTERN = /^[A-Za-z0-9_.-]+$/;

export function isValidKey(key) {
  return typeof key === 'string' && KEY_PATTERN.test(key);
}

function normalizeEntry(entry) {
  if (entry === null || typeof entry !== 'object' || Array.isArray(entry)) {
    return null;
  }
  return {
    value: typeof entry.value === 'string' ? entry.value : String(entry.value ?? ''),
    expiresAt: typeof entry.expiresAt === 'number' && Number.isFinite(entry.expiresAt)
      ? entry.expiresAt
      : null,
  };
}

export function loadStore(file) {
  let raw;
  try {
    raw = fs.readFileSync(file, 'utf8');
  } catch (err) {
    if (err && err.code === 'ENOENT') return Object.create(null);
    throw err;
  }
  if (raw.trim() === '') return Object.create(null);

  let parsed;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new Error(`invalid store file: ${file}`);
  }
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`invalid store file: ${file}`);
  }

  const data = Object.create(null);
  for (const key of Object.keys(parsed)) {
    const entry = normalizeEntry(parsed[key]);
    if (entry) data[key] = entry;
  }
  return data;
}

export function purgeExpired(data, now = Date.now()) {
  let changed = false;
  for (const key of Object.keys(data)) {
    const entry = data[key];
    if (entry && typeof entry.expiresAt === 'number' && entry.expiresAt <= now) {
      delete data[key];
      changed = true;
    }
  }
  return changed;
}

export function saveStore(file, data) {
  const absolute = path.resolve(file);
  const dir = path.dirname(absolute);
  fs.mkdirSync(dir, { recursive: true });
  const base = path.basename(absolute);
  const tmp = path.join(dir, `.${base}.${process.pid}.${randomBytes(6).toString('hex')}.tmp`);
  const json = `${JSON.stringify(data, null, 2)}\n`;
  fs.writeFileSync(tmp, json, 'utf8');
  try {
    fs.renameSync(tmp, absolute);
  } catch (err) {
    try { fs.unlinkSync(tmp); } catch { /* ignore */ }
    throw err;
  }
}
