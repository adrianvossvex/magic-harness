```js
function loadStore(file) {
  let raw;
  try {
    raw = fs.readFileSync(file, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') return {};
    throw err;
  }
  if (raw.trim() === '') return {};
  const parsed = JSON.parse(raw);
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error('store file is not a JSON object');
  }
  return parsed;
}

function saveStore(file, data) {
  const dir = path.dirname(path.resolve(file));
  const tmp = path.join(dir, `.${path.basename(file)}.${process.pid}.${Date.now()}.${Math.random().toString(36).slice(2)}.tmp`);
  try {
    fs.writeFileSync(tmp, JSON.stringify(data, null, 2) + '\n', 'utf8');
    fs.renameSync(tmp, file);
  } catch (err) {
    try { fs.unlinkSync(tmp); } catch {}
    throw err;
  }
}
```

```js
      if (!Number.isFinite(seconds) || seconds < 0) throw new Error(`invalid ttl: ${argv[i]}`);
      ttl = seconds;
    } else if (arg.startsWith('--')) {
      throw new Error(`unknown option: ${arg}`);
    } else {
      positional.push(arg);
    }
  }
  return { positional, file, ttl };
}
```

```js
export function run(argv) {
  let parsed;
  try {
    parsed = parseArgs(argv);
  } catch (err) {
    process.stderr.write(`${err.message}\n\n${USAGE}\n`);
    return 2;
  }

  const { positional, file, ttl } = parsed;
  const command = positional[0];

  if (!command) {
    process.stderr.write(`${USAGE}\n`);
    return 2;
  }

  try {
    switch (command) {
      case 'set':
        return commandSet(positional, file, ttl);
      case 'get':
        return commandGet(positional, file);
      case 'del':
        return commandDel(positional, file);
      case 'list':
        return commandList(positional, file);
      default:
        process.stderr.write(`Unknown command: ${command}\n\n${USAGE}\n`);
        return 2;
    }
  } catch (err) {
    process.stderr.write(`error: ${err.message}\n`);
    return 1;
  }
}
```

```js
function commandSet(positional, file, ttl) {
  const key = positional[1];
  if (key === undefined) {
    process.stderr.write(`${USAGE}\n`);
    return 2;
  }
  if (!KEY_PATTERN.test(key)) {
    process.stderr.write(`invalid key: ${key}\n`);
    return 1;
  }
  const values = positional.slice(2);
  if (values.length === 0) {
    process.stderr.write(`${USAGE}\n`);
    return 2;
  }
  const value = values.join(' ');
  const data = loadStore(file);
  const entry = { value };
  if (ttl !== null) entry.expiresAt = Date.now() + ttl * 1000;
  data[key] = entry;
  saveStore(file, data);
  return 0;
}
```

commandGet:
```js
function commandGet(positional, file) {
  const key = positional[1];
  if (key === undefined) { usage; return 2; }
  if (!KEY_PATTERN.test(key)) { stderr; return 1; }
  const data = loadStore(file);
  const now = Date.now();
  const entry = data[key];
  if (entry === undefined) {
    process.stderr.write(`not found: ${key}\n`);
    return 1;
  }
  if (isExpired(entry, now)) {
    delete data[key];
    saveStore(file, data);
    process.stderr.write(`not found: ${key}\n`);
    return 1;
  }
  process.stdout.write(`${String(entry.value)}\n`);
  return 0;
}
```

loadStore with validation:
```js
function loadStore(file) {
  let raw;
  try {
    raw = fs.readFileSync(file, 'utf8');
  } catch (err) {
    if (err && err.code === 'ENOENT') return {};
    throw err;
  }
  if (raw.trim() === '') return {};
  let parsed;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new Error(`store file is not valid JSON: ${file}`);
  }
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`store file must contain a JSON object: ${file}`);
  }
  return parsed;
}
```

```js
function saveStore(file, data) {
  const target = path.resolve(file);
  const dir = path.dirname(target);
  const tmp = path.join(
    dir,
    `.${path.basename(target)}.${process.pid}.${Date.now()}.${Math.random().toString(36).slice(2)}.tmp`,
  );
  fs.writeFileSync(tmp, `${JSON.stringify(data, null, 2)}\n`, 'utf8');
  try {
    fs.renameSync(tmp, target);
  } catch (err) {
    try { fs.unlinkSync(tmp); } catch { /* ignore */ }
    throw err;
  }
}
```
