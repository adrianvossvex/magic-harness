```js
export function readData(file = currentDataFile) {
  if (!existsSync(file)) {
    const data = emptyData();
    writeData(data, file);
    return data;
  }
  const raw = readFileSync(file, 'utf8').trim();
  if (raw === '') {
    const data = emptyData();
    writeData(data, file);
    return data;

  // normalize
  return {
    books: parsed.books ?? [],
    members: parsed.members ?? [],
    loans: parsed.loans ?? [],
    counters: parsed.counters ?? {},
  };
}
```

```js
export function checkout({ bookId, memberId, days = 14, today, data: dataFile } = {}) {
  const file = dataFile ?? getDataFile();
  const data = readData(file);
  const bid = toInt(bookId, 'book id');
  const mid = toInt(memberId, 'member id');
  const nDays = toInt(days, 'days');
  if (nDays < 1) throw new LibraryError('days must be a positive integer');
  const book = data.books.find(b => b.id === bid);
  if (!book) throw new LibraryError(`book ${bid} not found`);
  const member = data.members.find(m => m.id === mid);
  if (!member) throw new LibraryError(`member ${mid} not found`);
  if (book.available < 1) throw new LibraryError(`no available copies of book ${bid}`);
  const active = data.loans.filter(l => l.memberId === mid && l.returned === null).length;
  if (active >= 3) throw new LibraryError(`member ${mid} already has 3 active loans`);
  const out = today ?? todayISO();
  parseDate(out); // validate
  const due = addDays(out, nDays);
  const loan = { id: nextId(data,'loan'), bookId: bid, memberId: mid, out, due, returned: null, fine: 0 };
  book.available -= 1;
  data.loans.push(loan);
  writeData(data, file);
  return loan;
}
```

```js
function parse(argv) {
  const command = argv[0];
  if (!command || command.startsWith('-')) throw new UsageError(...);
  const spec = SPECS[command];
  if (!spec) throw new UsageError(`unknown command: ${command}`);
  const opts = {};
  const rest = argv.slice(1);
  let data;
  for (let i=0;i<rest.length;i++){
    const arg = rest[i];
    if (!arg.startsWith('--')) throw new UsageError(`unexpected argument: ${arg}`);
    let key, inline;
    const eq = arg.indexOf('=');
    if (eq !== -1) { key = arg.slice(2, eq); inline = arg.slice(eq+1); }
    else key = arg.slice(2);
    if (key === 'data') {
      data = inline !== undefined ? inline : rest[++i];
      if (data === undefined) throw new UsageError('missing value for --data');
      continue;
    }
    const type = spec[key];
    if (!type) throw new UsageError(`unknown option: --${key}`);
    if (type === 'bool') {
      if (inline !== undefined) throw new UsageError(`option --${key} takes no value`);
      opts[key] = true;
    } else {
      const val = inline !== undefined ? inline : rest[++i];
      if (val === undefined || val.startsWith('--')) throw new UsageError(`missing value for --${key}`);
      if (type === 'int') {
        if (!/^-?\d+$/.test(val)) throw new UsageError(`invalid value for --${key}: ${val}`);
        opts[key] = Number(val);
      } else opts[key] = val;
    }
  }
  return { command, opts, data };
}
```
