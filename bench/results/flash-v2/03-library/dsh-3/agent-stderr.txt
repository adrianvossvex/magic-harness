```js
import fs from 'node:fs';
import path from 'node:path';

export const DEFAULT_DATA_PATH = 'data/library.json';

export function resolveDataPath(dataPath) {
  return dataPath || process.env.LIBRARY_DATA || DEFAULT_DATA_PATH;
}

export function emptyData() { return { books: [], members: [], loans: [] }; }

export function load(dataPath) {
  const file = resolveDataPath(dataPath);
  if (!fs.existsSync(file)) {
    // create on first use
    save(file, emptyData());
    return emptyData();
  }
  const raw = fs.readFileSync(file, 'utf8');
  if (!raw.trim()) return emptyData();
  let parsed;
  try { parsed = JSON.parse(raw); }
  catch { throw new Error(`invalid data file: ${file}`); }
  if (!parsed || typeof parsed !== 'object') throw new Error(...);
  return {
    books: Array.isArray(parsed.books) ? parsed.books : [],
    members: Array.isArray(parsed.members) ? parsed.members : [],
    loans: Array.isArray(parsed.loans) ? parsed.loans : [],
  };
}

export function save(dataPath, data) {
  const file = resolveDataPath(dataPath);
  fs.mkdirSync(path.dirname(path.resolve(file)), { recursive: true });
  fs.writeFileSync(file, JSON.stringify(data, null, 2) + '\n');
}

export function nextId(items) {
  return items.reduce((max, item) => (item && Number.isInteger(item.id) && item.id > max ? item.id : max), 0) + 1;
}
```
