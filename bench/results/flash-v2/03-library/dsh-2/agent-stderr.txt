```js
function parseArgs(argv, spec) {
  const opts = {};
  const flags = {};
  for (let i = 0; i < argv.length; i++) {
    let token = argv[i];
    if (!token.startsWith('--')) throw new UsageError(`unexpected argument: ${token}`);
    let name = token.slice(2);
    let inlineValue = null;
    const eq = name.indexOf('=');
    if (eq !== -1) { inlineValue = name.slice(eq+1); name = name.slice(0, eq); }
    if (!(name in spec)) throw new UsageError(`unknown option: --${name}`);
    if (spec[name] === 'boolean') {
      if (inlineValue !== null) throw new UsageError(`option --${name} does not take a value`);
      opts[name] = true;
    } else {
      let value;
      if (inlineValue !== null) value = inlineValue;
      else {
        if (i+1 >= argv.length || argv[i+1].startsWith('--')) throw new UsageError(`missing value for --${name}`);
        value = argv[++i];
      }
      opts[name] = value;
    }
  }
  return opts;
}
```

```js
import { LibraryError } from './errors.js';
import { readState, update } from './storage.js';

function requireNonEmptyString(value, label) {
  if (typeof value !== 'string' || value.trim() === '') {
    throw new LibraryError(`${label} is required`);
  }
  return value.trim();
}

function requirePositiveInteger(value, label) {
  if (!Number.isInteger(value) || value < 1) {
    throw new LibraryError(`${label} must be a positive integer`);
  }
  return value;
}

export function addBook({ title, author, copies = 1 } = {}) {
  const cleanTitle = requireNonEmptyString(title, 'title');
  const cleanAuthor = requireNonEmptyString(author, 'author');
  const cleanCopies = requirePositiveInteger(copies, 'copies');
  return update((state) => {
    const book = { id: state.nextIds.book, title: cleanTitle, author: cleanAuthor, copies: cleanCopies, available: cleanCopies };
    state.nextIds.book += 1;
    state.books.push(book);
    return { ...book };
  });
}

export function listBooks({ available } = {}) {
  const state = readState();
  const books = state.books.map((b) => ({ ...b }));
  if (available) return books.filter((b) => b.available > 0);
  return books;
}
```

```js
import { LibraryError } from './errors.js';

const DATE_RE = /^(\d{4})-(\d{2})-(\d{2})$/;

export function currentDate(now = new Date()) {
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, '0');
  const d = String(now.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

export function parseDate(value, label = 'date') {
  if (typeof value !== 'string') throw new LibraryError(`invalid ${label}: ${value}`);
  const match = DATE_RE.exec(value);
  if (!match) throw new LibraryError(`invalid ${label}: ${value}`);
  const year = Number(match[1]); const month = Number(match[2]); const day = Number(match[3]);
  const ms = Date.UTC(year, month - 1, day);
  const dt = new Date(ms);
  if (dt.getUTCFullYear() !== year || dt.getUTCMonth() !== month - 1 || dt.getUTCDate() !== day) {
    throw new LibraryError(`invalid ${label}: ${value}`);
  }
  return ms;
}

export function formatDate(ms) {
  const dt = new Date(ms);
  const y = dt.getUTCFullYear();
  const m = String(dt.getUTCMonth() + 1).padStart(2, '0');
  const d = String(dt.getUTCDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

export function addDays(dateStr, days) {
  return formatDate(parseDate(dateStr) + days * 86400000);
}

export function daysBetween(from, to) {
  return Math.round((parseDate(to) - parseDate(from)) / 86400000);
}

export function resolveToday(today) {
  if (today === undefined || today === null) return currentDate();
  parseDate(today, 'today');
  return today;
}
```

```js
import { LibraryError } from './errors.js';
import { addDays, daysBetween, resolveToday } from './dates.js';
import { readState, update } from './storage.js';

const MAX_ACTIVE_LOANS = 3;
const FINE_PER_DAY = 0.5;
const MAX_FINE = 20;

function requirePositiveInteger(value, label) {
  if (!Number.isInteger(value) || value < 1) throw new LibraryError(`${label} must be a positive integer`);
  return value;
}

function findBook(state, bookId) {
  const book = state.books.find((b) => b.id === bookId);
  if (!book) throw new LibraryError(`book ${bookId} not found`);
  return book;
}
function findMember(state, memberId) {
  const member = state.members.find((m) => m.id === memberId);
  if (!member) throw new LibraryError(`member ${memberId} not found`);
  return member;
}

export function fineFor(due, returned) {
  const late = daysBetween(due, returned);
  if (late <= 0) return 0;
  return Math.round(Math.min(late * FINE_PER_DAY, MAX_FINE) * 100) / 100;
}

export function checkout({ bookId, memberId, days = 14, today } = {}) {
  const bookIdInt = requirePositiveInteger(bookId, 'bookId');
  const memberIdInt = requirePositiveInteger(memberId, 'memberId');
  const daysInt = requirePositiveInteger(days, 'days');
  const out = resolveToday(today);
  return update((state) => {
    const book = findBook(state, bookIdInt);
    findMember(state, memberIdInt);
    if (book.available < 1) throw new LibraryError(`no available copies of book ${bookIdInt}`);
    const active = state.loans.filter((l) => l.memberId === memberIdInt && l.returned === null).length;
    if (active >= MAX_ACTIVE_LOANS) throw new LibraryError(`member ${memberIdInt} already has ${MAX_ACTIVE_LOANS} active loans`);
    const loan = { id: state.nextIds.loan, bookId: bookIdInt, memberId: memberIdInt, out, due: addDays(out, daysInt), returned: null, fine: 0 };
    state.nextIds.loan += 1;
    book.available -= 1;
    state.loans.push(loan);
    return { ...loan };
  });
}

export function returnLoan({ loanId, today } = {}) {
  const loanIdInt = requirePositiveInteger(loanId, 'loanId');
  const returnedOn = resolveToday(today);
  return update((state) => {
    const loan = state.loans.find((l) => l.id === loanIdInt);
    if (!loan) throw new LibraryError(`loan ${loanIdInt} not found`);
    if (loan.returned !== null) throw new LibraryError(`loan ${loanIdInt} has already been returned`);
    loan.returned = returnedOn;
    loan.fine = fineFor(loan.due, returnedOn);
    const book = state.books.find((b) => b.id === loan.bookId);
    if (book) book.available += 1;
    return { ...loan };
  });
}

export function listLoans({ memberId, overdue, today } = {}) {
  const state = readState();
  const on = resolveToday(today);
  let loans = state.loans.map((l) => ({ ...l }));
  if (memberId !== undefined && memberId !== null) {
    const id = requirePositiveInteger(memberId, 'memberId');
    loans = loans.filter((l) => l.memberId === id);
  }
  if (overdue) loans = loans.filter((l) => l.returned === null && daysBetween(l.due, on) > 0);
  return loans;
}

export function report({ today } = {}) {
  const state = readState();
  const on = resolveToday(today);
  const books = state.books.length;
  const copies = state.books.reduce((s, b) => s + b.copies, 0);
  const available = state.books.reduce((s, b) => s + b.available, 0);
  const members = state.members.length;
  const activeLoans = state.loans.filter((l) => l.returned === null).length;
  const overdue = state.loans.filter((l) => l.returned === null && daysBetween(l.due, on) > 0).length;
  const finesOwed = state.loans.filter((l) => l.returned !== null).reduce((s, l) => s + (l.fine || 0), 0);
  return { books, copies, available, members, activeLoans, overdue, finesOwed: Math.round(finesOwed*100)/100 };
}
```

```js
function parseOptions(argv, spec) {
  const opts = {};
  for (let i = 0; i < argv.length; i++) {
    const token = argv[i];
    if (!token.startsWith('--') || token === '--') throw new UsageError(`unexpected argument: ${token}`);
    let name = token.slice(2);
    let inline;
    const eq = name.indexOf('=');
    if (eq !== -1) { inline = name.slice(eq + 1); name = name.slice(0, eq); }
    if (!Object.prototype.hasOwnProperty.call(spec, name)) throw new UsageError(`unknown option: --${name}`);
    if (spec[name] === 'boolean') {
      if (inline !== undefined) throw new UsageError(`option --${name} does not take a value`);
      opts[name] = true;
    } else if (inline !== undefined) {
      opts[name] = inline;
    } else {
      const next = argv[i + 1];
      if (next === undefined || (next.startsWith('--') && next.length > 2)) throw new UsageError(`missing value for option: --${name}`);
      opts[name] = next; i++;
    }
  }
  return opts;
}
```
