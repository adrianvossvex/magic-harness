import { readFileSync, writeFileSync, renameSync, existsSync, mkdirSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { randomBytes } from 'node:crypto';

export function readStore(filePath) {
  let raw;
  try {
    raw = readFileSync(filePath, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') return {};
    throw err;
  }
  if (raw.trim() === '') return {};
  let parsed;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new Error(`invalid store file: ${filePath}`);
  }
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`invalid store file: ${filePath}`);
  }
  return parsed;
}

export function writeStore(filePath, store) {
  const dir = dirname(filePath);
  const tmp = resolve(dir, `.${basename?}...`);
  writeFileSync(tmp, JSON.stringify(store, null, 2) + '\n');
  renameSync(tmp, filePath);
}

#!/usr/bin/env node
import process from 'node:process';
import { resolve } from 'node:path';
import { readStore, writeStore, isExpired, KEY_RE } from '../lib/store.js';

const USAGE = `Usage: kv <command> [args] [--file <path>]

Commands:
  set <key> <value...> [--ttl <seconds>]   Store a value (optionally expiring)
  get <key>                                Print a value
  del <key>                                Delete a key
  list                                     List non-expired keys

Options:
  --file <path>   Store file (default: .kv.json)
  --ttl <seconds> Expiry for set (absolute ms stored in file)
`;

function fail(message, code) {
  process.stderr.write(message + '\n');
  process.exit(code);
}

function parseArgs(argv) {
  const flags = { file: null, ttl: null };
  const positionals = [];
  for (let i = 0; i < argv.length; i++) {
    const arg = argv[i];
    if (arg === '--file') {
      if (i + 1 >= argv.length) fail('missing value for --file', 2);
      flags.file = argv[++i];
    } else if (arg === '--ttl') {
      if (i + 1 >= argv.length) fail('missing value for --ttl', 2);
      flags.ttl = argv[++i];
    } else if (arg.startsWith('--file=')) {
      flags.file = arg.slice('--file='.length);
    } else if (arg.startsWith('--ttl=')) {
      flags.ttl = arg.slice('--ttl='.length);
    } else {
      positionals.push(arg);
    }
  }
  return { flags, positionals };
}
