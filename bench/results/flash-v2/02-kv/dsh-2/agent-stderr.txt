```js
import { promises as fs } from 'node:fs';
import path from 'node:path';

export async function loadStore(file) {
  let raw;
  try {
    raw = await fs.readFile(file, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') return {};
    throw err;
  }
  if (raw.trim() === '') return {};
  const parsed = JSON.parse(raw);
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`invalid store file: ${file}`);
  }
  return parsed;
}

export async function saveStore(file, store) {
  const dir = path.dirname(file);

  await fs.writeFile(tmp, data, 'utf8');

}
```

lib/store.js:

```js
import { promises as fs } from 'node:fs';
import path from 'node:path';
import process from 'node:process';

export function nowMs() {
  return Date.now();
}

export function isExpired(entry, now = Date.now()) {
  return (
    entry != null &&
    typeof entry === 'object' &&
    typeof entry.expiresAt === 'number' &&
    entry.expiresAt <= now
  );
}

export function pruneExpired(store, now = Date.now()) {
  let removed = false;
  for (const key of Object.keys(store)) {
    if (isExpired(store[key], now)) {
      delete store[key];
      removed = true;
    }
  }
  return removed;
}

export async function loadStore(file) {
  let raw;
  try {
    raw = await fs.readFile(file, 'utf8');
  } catch (err) {
    if (err.code === 'ENOENT') return {};
    throw err;
  }
  if (raw.trim() === '') return {};
  let parsed;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new Error(`store file is not valid JSON: ${file}`);
  }
  if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
    throw new Error(`store file must contain a JSON object: ${file}`);
  }
  return parsed;
}

export async function saveStore(file, store) {
  const dir = path.dirname(file);
  await fs.mkdir(dir, { recursive: true });
  const base = path.basename(file);
  const tmp = path.join(dir, `.${base}.${process.pid}.${Date.now()}.${Math.random().toString(36).slice(2)}.tmp`);
  const data = `${JSON.stringify(store, null, 2)}\n`;
  try {
    await fs.writeFile(tmp, data, 'utf8');
    await fs.rename(tmp, file);
  } catch (err) {
    await fs.rm(tmp, { force: true }).catch(() => {});
    throw err;
  }
}
```
