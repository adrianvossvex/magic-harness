// A tiny in-memory todo list.
export function createStore() {
  const items = [];
  return {
    add(text) { const item = { id: items.length + 1, text, done: false }; items.push(item); return item; },
    list() { return items.slice(); },
    toggle(id) { const item = items.find((i) => i.id === id); if (!item) throw new Error('Unknown id: ' + id); item.done = !item.done; return item; },
    remove(id) { const index = items.findIndex((i) => i.id === id); if (index === -1) return false; items.splice(index, 1); return true; },
    count() { return items.length; },
  };
}
