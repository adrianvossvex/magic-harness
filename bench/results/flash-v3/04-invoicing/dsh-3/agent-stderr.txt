export function setDataPath(p) { if (p) currentDataPath = p; }

export function emptyDb() {
  return {
    customers: [],
    products: [],
    invoices: [],

  };

export function normalizeDb(raw) { ... merge with emptyDb ... }
export function loadDb(path = currentDataPath) {
  const abs = path;
  if (!fs.existsSync(abs)) {
    const db = emptyDb();
    saveDb(db, abs);
    return db;
  }
  const text = fs.readFileSync(abs, "utf8");
  if (!text.trim()) return emptyDb();
  let parsed;
  try { parsed = JSON.parse(text); } catch { throw new Error(`Invalid data file: ${path}`); }
  return normalizeDb(parsed);
}
export function saveDb(db, path = currentDataPath) {
  fs.mkdirSync(path.dirname(path), { recursive: true });
  fs.writeFileSync(path, JSON.stringify(db, null, 2) + "\n");
