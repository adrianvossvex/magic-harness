```js
function parseOptions(args, schema) {
  const values = {};
  const booleans = schema.booleans || [];
  const allowed = new Set([...(schema.options||[]), 'data', ...booleans]);
  let i=0;
  while (i<args.length) {
    const tok = args[i];
    if (!tok.startsWith('--')) throw new UsageError(`unexpected argument: ${tok}`);
    const name = tok.slice(2);
    if (!allowed.has(name)) throw new UsageError(`unknown option: --${name}`);
    if (booleans.includes(name)) { values[name]=true; i+=1; }
    else {
      const v = args[i+1];
      if (v === undefined || (v.startsWith('--') && v.length>2)) throw new UsageError(`missing value for --${name}`);
      values[name]=v; i+=2;
    }
  }
  for (const req of (schema.required||[])) if (values[req]===undefined) throw new UsageError(`missing required option: --${req}`);
  return values;
}
```

// Money helpers. Amounts are integer cents; external amounts are decimal
// strings with at most two decimal places.

const AMOUNT_RE = /^\d+(?:\.\d{1,2})?$/;

export function parseMoney(text) {
  if (typeof text !== "string" || !AMOUNT_RE.test(text)) {
    throw new Error(`invalid amount: ${String(text)}`);
  }
  const [whole, fraction = ""] = text.split(".");
  const cents = Number(whole) * 100 + Number((fraction + "00").slice(0, 2));
  return cents;
}

export function formatMoney(cents) {
  if (!Number.isInteger(cents)) {
    throw new Error(`invalid cents: ${String(cents)}`);
  }
  const sign = cents < 0 ? "-" : "";
  const abs = Math.abs(cents);
  const whole = Math.floor(abs / 100);
  const fraction = abs % 100;
  return `${sign}${whole}.${String(fraction).padStart(2, "0")}`;
}

export function roundCents(value) {
  if (typeof value !== "number" || !Number.isFinite(value)) {
    throw new Error(`invalid number: ${String(value)}`);
  }
  return value < 0 ? -Math.floor(-value + 0.5) : Math.floor(value + 0.5);
}

import fs from "node:fs";
import path from "node:path";

export const DEFAULT_DATA_PATH = "data/invoicing.json";

let state = null;
let activePath = null;

function emptyDb() {
  return {
    customers: [],
    products: [],
    invoices: [],
    counters: { customer: 0, product: 0, invoice: 0 },
  };
}

function normalize(raw) {
  const db = emptyDb();
  if (raw && typeof raw === "object") {
    if (Array.isArray(raw.customers)) db.customers = raw.customers;
    if (Array.isArray(raw.products)) db.products = raw.products;
    if (Array.isArray(raw.invoices)) db.invoices = raw.invoices;
    if (raw.counters && typeof raw.counters === "object") {
      for (const key of ["customer", "product", "invoice"]) {
        const n = raw.counters[key];
        if (Number.isInteger(n) && n >= 0) db.counters[key] = n;
      }
    }
  }
  return db;
}

export function resolveDataPath(dataPath) {
  return path.resolve(process.cwd(), dataPath || DEFAULT_DATA_PATH);
}

export function useData(dataPath) {
  activePath = resolveDataPath(dataPath);
  if (fs.existsSync(activePath)) {
    const text = fs.readFileSync(activePath, "utf8");
    let raw;
    try {
      raw = JSON.parse(text);
    } catch {
      throw new Error(`invalid JSON in data file: ${activePath}`);
    }
    state = normalize(raw);
  } else {
    state = emptyDb();
    save();
  }
  return state;
}

export function getData() {
  if (state === null) useData();
  return state;
}

export function save() {
  if (state === null) return;
  if (activePath === null) activePath = resolveDataPath();
  fs.mkdirSync(path.dirname(activePath), { recursive: true });
  fs.writeFileSync(activePath, `${JSON.stringify(state, null, 2)}\n`);
}

export function resetData() {
  state = null;
  activePath = null;
}

export function nextId(entity) {
  const db = getData();
  db.counters[entity] = (db.counters[entity] || 0) + 1;
  return db.counters[entity];
}

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

export function currentDate(now = new Date()) {
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, "0");
  const d = String(now.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

export function normalizeDate(value) {
  if (typeof value !== "string" || !DATE_RE.test(value)) {
    throw new Error(`invalid date: ${String(value)}`);
  }
  const [y, m, d] = value.split("-").map(Number);
  const dt = new Date(Date.UTC(y, m - 1, d));
  if (dt.getUTCFullYear() !== y || dt.getUTCMonth() !== m - 1 || dt.getUTCDate() !== d) {
    throw new Error(`invalid date: ${value}`);
  }
  return value;
}

function dateValue(date) {
  const [y, m, d] = date.split("-").map(Number);
  return Date.UTC(y, m - 1, d);
}

function addDays(date, days) {
  const dt = new Date(dateValue(date) + days * 86400000);
  const y = dt.getUTCFullYear();
  const m = String(dt.getUTCMonth() + 1).padStart(2, "0");
  const d = String(dt.getUTCDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

function daysBetween(from, to) {
  return Math.round((dateValue(to) - dateValue(from)) / 86400000);
}

  if (min !== null && n < min) throw new Error(`${label} must be at least ${min}`);
  if (max !== null && n > max) throw new Error(`${label} must be at most ${max}`);

export function addLine({ invoiceId, productId, quantity = 1, discount = 0 } = {}) {
  const invoice = findInvoice(invoiceId);
  if (invoice.status !== "draft") throw new Error("only draft invoices can accept lines");
  const db = getData();
  const pid = requireInteger(productId, "productId", { min: 1 });
  const product = db.products.find((p) => p.id === pid);
  if (!product) throw new Error(`product ${pid} not found`);
  const qty = requireInteger(quantity, "quantity", { min: 1 });
  const disc = typeof discount === "string" && discount.trim() !== "" ? Number(discount) : discount;
  if (typeof disc !== "number" || !Number.isFinite(disc) || disc < 0 || disc > 100) {
    throw new Error("discount must be between 0 and 100");
  }
  const net = roundCents((product.price * qty * (100 - disc)) / 100);
  const tax = roundCents((net * product.taxRate) / 100);
  invoice.lines.push({
    productId: product.id,
    description: product.name,
    quantity: qty,
    unitPrice: product.price,
    discount: disc,
    net,
    tax,
  });

  return invoice;

export function issue({ invoiceId, today } = {}) {
  const invoice = findInvoice(invoiceId);
  if (invoice.status !== "draft") throw new Error("only draft invoices can be issued");
  if (invoice.lines.length === 0) throw new Error("cannot issue an invoice without lines");
  const date = normalizeDate(today ?? currentDate());
  const year = date.slice(0, 4);
  let highest = 0;
  for (const other of getData().invoices) {
    const match = /^INV-(\d{4})-(\d{4,})$/.exec(other.number || "");
    if (match && match[1] === year) highest = Math.max(highest, Number(match[2]));
  }
  invoice.number = `INV-${year}-${String(highest + 1).padStart(4, "0")}`;
  invoice.status = "issued";
  invoice.issued = date;
  invoice.due = addDays(date, invoice.terms);
  save();
  return invoice;
}

export function pay({ invoiceId, amount, today } = {}) {
  const invoice = findInvoice(invoiceId);
  if (invoice.status !== "issued") throw new Error("only issued invoices can be paid");
  let cents;
  if (typeof amount === "number") {
    if (!Number.isInteger(amount)) throw new Error("amount must be an integer number of cents");

import { getData } from "./store.js";
import { currentDate, normalizeDate, daysBetween } from "./invoices.js";

export function aging({ today } = {}) {
  const reference = normalizeDate(today ?? currentDate());
  const buckets = { current: 0, days1to30: 0, days31to60: 0, days61to90: 0, over90: 0 };
  for (const invoice of getData().invoices) {
    if (invoice.status !== "issued") continue;
    const overdueDays = daysBetween(invoice.due, reference);
    if (overdueDays <= 0) buckets.current += invoice.balance;
    else if (overdueDays <= 30) buckets.days1to30 += invoice.balance;
    else if (overdueDays <= 60) buckets.days31to60 += invoice.balance;
    else if (overdueDays <= 90) buckets.days61to90 += invoice.balance;
    else buckets.over90 += invoice.balance;
  }
  const total = buckets.current + buckets.days1to30 + buckets.days31to60 + buckets.days61to90 + buckets.over90;
  return { ...buckets, total };
}

export function revenue() {
  const months = new Map();
  for (const invoice of getData().invoices) {
    if (invoice.status !== "issued" && invoice.status !== "paid") continue;
    if (!invoice.issued) continue;
    const month = invoice.issued.slice(0, 7);
    const entry = months.get(month) || { month, invoices: 0, total: 0 };
    entry.invoices += 1;
    entry.total += invoice.total;
    months.set(month, entry);
  }
  return [...months.values()].sort((a, b) => (a.month < b.month ? -1 : a.month > b.month ? 1 : 0));
}

function parseOptions(args, schema) {
  const values = {};
  const allowed = new Set([...(schema.options || []), ...(schema.booleans || []), "data"]);
  const booleans = new Set(schema.booleans || []);
  let i = 0;
  while (i < args.length) {
    const token = args[i];
    if (!token.startsWith("--")) throw new UsageError(`unexpected argument: ${token}`);
    const name = token.slice(2);
    if (!allowed.has(name)) throw new UsageError(`unknown option: ${token}`);
    if (booleans.has(name)) {
      values[name] = true;
      i += 1;
      continue;
    }
    const value = args[i + 1];
    if (value === undefined || (value.startsWith("--") && value.length > 2)) {
      throw new UsageError(`missing value for ${token}`);
    }
    values[name] = value;
    i += 2;

  for (const required of schema.required || []) {

    rest = argv.slice(1);
  }
  if (!command) {

    process.exit(1);

    if (!schema) throw new UsageError(`unknown command: ${command}`);

    useData(options.data);
    const result = run(command, options);

function extractDataPath(argv) {
  const rest = [];

  for (let i=0;i<argv.length;i++){
    const token = argv[i];
    if (token === "--data") {
      const value = argv[i+1];
