```js
const MONEY_RE = /^\d+(?:\.\d{1,2})?$/;

function error(message) {
  throw new Error(message);
}

export function parseMoney(text) {
  let value = text;
  if (typeof value === 'number') {
    if (!Number.isFinite(value)) error(`Invalid money amount: ${text}`);
    value = String(value);
  }
  if (typeof value !== 'string' || !MONEY_RE.test(value)) {
    error(`Invalid money amount: ${text}`);
  }
  const [whole, fraction = ''] = value.split('.');
  const padded = (fraction + '00').slice(0, 2);
  return Number.parseInt(whole, 10) * 100 + Number.parseInt(padded, 10);

export function formatMoney(cents) {
```

```js
export function isValidDate(text) {
  if (typeof text !== 'string' || !DATE_RE.test(text)) return false;
  const [year, month, day] = text.split('-').map(Number);
  if (month < 1 || month > 12 || day < 1 || day > 31) return false;
  const date = new Date(Date.UTC(year, month - 1, day));
  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCDate() === day
  );
}

export function assertDate(text, label = 'date') {
  if (!isValidDate(text)) throw new Error(`Invalid ${label}: ${text}`);
  return text;
}

export function todayISO() {
  const now = new Date();
  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const day = String(now.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

export function toTimestamp(dateText) {
  assertDate(dateText);
  const [year, month, day] = dateText.split('-').map(Number);
  return Date.UTC(year, month - 1, day);
}

export function addDays(dateText, days) {
  assertDate(dateText);
  if (!Number.isInteger(days)) throw new Error(`Invalid number of days: ${days}`);
  const result = new Date(toTimestamp(dateText) + days * 86400000);
  const year = result.getUTCFullYear();
  const month = String(result.getUTCMonth() + 1).padStart(2, '0');
  const day = String(result.getUTCDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}
```

```js
export function aging({ today } = {}) {
  const reference = today === undefined ? todayISO() : assertDate(today, 'today');
  const buckets = {
    current: 0,
    days1to30: 0,
    days31to60: 0,
    days61to90: 0,
    over90: 0,
    total: 0,
  };
  for (const invoice of loadDatabase().invoices) {
    if (invoice.status !== 'issued') continue;
    const daysPastDue = daysBetween(invoice.due, reference);
    if (daysPastDue <= 0) buckets.current += invoice.balance;
    else if (daysPastDue <= 30) buckets.days1to30 += invoice.balance;
    else if (daysPastDue <= 60) buckets.days31to60 += invoice.balance;
    else if (daysPastDue <= 90) buckets.days61to90 += invoice.balance;
    else buckets.over90 += invoice.balance;
  }
  buckets.total =
    buckets.current +
    buckets.days1to30 +
    buckets.days31to60 +
    buckets.days61to90 +
    buckets.over90;
  return buckets;
}

export function revenue() {
  const months = new Map();
  for (const invoice of loadDatabase().invoices) {
    if (invoice.status !== 'issued' && invoice.status !== 'paid') continue;
    if (!invoice.issued) continue;
    const month = invoice.issued.slice(0, 7);
    let entry = months.get(month);
    if (!entry) {
      entry = { month, invoices: 0, total: 0 };
      months.set(month, entry);
    }
    entry.invoices += 1;
    entry.total += invoice.total;
  }
  return [...months.values()].sort((a, b) => (a.month < b.month ? -1 : a.month > b.month ? 1 : 0));
}
```

```js
function parseArgs(argv) {
  const positionals = [];
  const options = {};
  for (let i = 0; i < argv.length; i += 1) {
    const arg = argv[i];
    if (arg.startsWith('--')) {
      const key = arg.slice(2);
      if (!KNOWN_OPTIONS.has(key)) throw new UsageError(`Unknown option: ${arg}`);
      if (FLAG_OPTIONS.has(key)) {

        continue;

      const next = argv[i + 1];
      if (next === undefined || next.startsWith('--')) {
        throw new UsageError(`Option ${arg} requires a value`);
      }
      options[key] = next;
      i += 1;

      positionals.push(arg);
```
