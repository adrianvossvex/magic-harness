```js
// A tiny in-memory todo list.
export function createStore() {
  const items = [];
  let nextId = 1;
  const find = (id) => items.find((item) => item.id === id);
  return {
    add(text) { const item = { id: nextId++, text, done: false }; items.push(item); return item; },
    list() { return items.slice(); },
    toggle(id) { const item = find(id); if (!item) throw new Error(`Unknown id: ${id}`); item.done = !item.done; return item; },
    remove(id) { const index = items.findIndex((item) => item.id === id); if (index === -1) return false; items.splice(index, 1); return true; },
    count() { return items.length; },
  };
}
```
