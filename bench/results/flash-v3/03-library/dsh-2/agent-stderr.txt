function parseArgs(argv) {
  const opts = {};
  const flags = new Set();
  for (let i=0;i<argv.length;i++){
    let tok = argv[i];
    if (!tok.startsWith('--')) throw usageError(`Unexpected argument: ${tok}`);
    let key, val;
    if (tok.includes('=')) { [key, ...rest] = tok.slice(2).split('='); val = rest.join('='); }
    else {
      key = tok.slice(2);
      // is it a known flag?
      if (FLAGS.has(key)) { flags.add(key); continue; }
      // otherwise require value
      if (i+1 >= argv.length || argv[i+1].startsWith('--')) throw usageError(`Missing value for --${key}`);

    opts[key]=val;
  }

}

function normalize(raw) {
  const state = emptyState();
  if (raw && typeof raw === 'object') {
    if (Array.isArray(raw.books)) state.books = raw.books;
    if (Array.isArray(raw.members)) state.members = raw.members;
    if (Array.isArray(raw.loans)) state.loans = raw.loans;
    if (raw.nextId && typeof raw.nextId === 'object') {

  // ensure counters at least max+1

    const next = Number(state.nextId[key]);
    if (!Number.isInteger(next) || next <= max) state.nextId[key] = max + 1;
  }
  return state;
}

  for (let i = 0; i < argv.length; i++) {

    if (typeof token === 'string' && token.startsWith('--')) {
      const body = token.slice(2);
      let key, value;

        value = body.slice(eq + 1);

      if (!VALUE_OPTIONS.has(key) && !FLAG_OPTIONS.has(key)) {
        throw new UsageError(`Unknown option: --${key}`);
      }
      if (FLAG_OPTIONS.has(key)) {
        if (value !== undefined) throw new UsageError(`Option --${key} does not take a value`);
        flags.add(key);
      } else {
        if (value === undefined) {
          if (i + 1 >= argv.length || argv[i+1].startsWith('--')) {
            throw new UsageError(`Missing value for --${key}`);
          }
          value = argv[++i];
        }
        opts[key] = value;
      }
    } else {

    }
  }
