import { readFileSync, writeFileSync, mkdirSync, existsSync } from 'node:fs';
import { dirname } from 'node:path';

export const DEFAULT_DATA_PATH = 'data/library.json';

let currentPath = DEFAULT_DATA_PATH;

export function setDataPath(p) {
  currentPath = p || DEFAULT_DATA_PATH;
  return currentPath;
}

export function getDataPath() { return currentPath; }

function emptyState() {
  return { sequences: { books: 0, members: 0, loans: 0 }, books: [], members: [], loans: [] };
}

export function readState() {
  if (!existsSync(currentPath)) {
    const state = emptyState();
    writeState(state); // create file on first use
    return state;
  }
  const raw = readFileSync(currentPath, 'utf8');
  if (raw.trim() === '') { ... empty }
  let parsed;
  try { parsed = JSON.parse(raw); } catch { throw new Error(...) }
  // normalize
  return { sequences: {...}, books: ..., ... };
}

export function writeState(state) {
  mkdirSync(dirname(currentPath), { recursive: true });
  writeFileSync(currentPath, JSON.stringify(state, null, 2) + '\n');
}

export function update(fn) {
  const state = readState();
  const result = fn(state);
  writeState(state);
  return result;
}

function parseArgs(argv, spec) {
  const opts = {};
  const positionals = [];
  for (let i = 0; i < argv.length; i++) {
    const token = argv[i];
    if (token === '--') { // rest positional
      positionals.push(...argv.slice(i+1)); break;
    }
    if (token.startsWith('--')) {
      let name = token.slice(2);
      let inline;
      const eq = name.indexOf('=');
      if (eq !== -1) { inline = name.slice(eq+1); name = name.slice(0, eq); }
      if (name === 'data') {
        if (inline !== undefined) { opts.data = inline; }
        else {
          const next = argv[++i];
          if (next === undefined || (next.startsWith('--') && next.length>2)) throw new UsageError('option --data requires a value');
          opts.data = next;
        }
        continue;
      }
      const def = spec.options.find(o => o.flag === name);
      if (!def) throw new UsageError(`unknown option --${name}`);
      if (def.type === 'boolean') {
        opts[def.key] = inline === undefined ? true : parseBool(inline, name);
        continue;
      }
      let value;
      if (inline !== undefined) value = inline;
      else {
        const next = argv[++i];
        if (next === undefined || (next.startsWith('--') && next.length>2)) throw new UsageError(`option --${name} requires a value`);
        value = next;
      }
      opts[def.key] = def.type === 'int' ? parsePositiveInt(value, name) : def.type === 'number' ? parsePositiveInt(value, name) : value;
    } else {
      positionals.push(token);
    }
  }
  // required
  for (const o of spec.options) {
    if (o.required && (opts[o.key] === undefined || opts[o.key] === '')) throw new UsageError(`missing required option --${o.flag}`);
  }
  return { opts, positionals };
}

import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { dirname, resolve } from 'node:path';
import { LibraryError } from './errors.js';

export const DEFAULT_DATA_PATH = 'data/library.json';

let configuredPath = null;

export function setDataPath(dataPath) {
  configuredPath = dataPath ?? null;
}

export function getDataPath() {
  return configuredPath || process.env.LIBRARY_DATA || DEFAULT_DATA_PATH;
}

export function resolveDataPath(dataPath) {
  return resolve(dataPath || getDataPath());
}

function emptyState() {
  return {
    sequences: { books: 0, members: 0, loans: 0 },
    books: [],
    members: [],
    loans: [],
  };
}

function normalize(value) {
  const base = emptyState();
  if (!value || typeof value !== 'object' || Array.isArray(value)) return base;
  const seq = value.sequences && typeof value.sequences === 'object' ? value.sequences : {};
  return {
    sequences: {
      books: Number.isInteger(seq.books) ? seq.books : 0,
      members: Number.isInteger(seq.members) ? seq.members : 0,
      loans: Number.isInteger(seq.loans) ? seq.loans : 0,
    },
    books: Array.isArray(value.books) ? value.books : [],
    members: Array.isArray(value.members) ? value.members : [],
    loans: Array.isArray(value.loans) ? value.loans : [],
  };
}

export function readState(dataPath) {
  const file = resolveDataPath(dataPath);
  if (!existsSync(file)) {
    const state = emptyState();
    writeState(dataPath, state);
    return state;
  }
  const raw = readFileSync(file, 'utf8');
  if (raw.trim() === '') return emptyState();
  let parsed;
  try {
    parsed = JSON.parse(raw);
  } catch {
    throw new LibraryError(`data file is not valid JSON: ${file}`);
  }
  return normalize(parsed);
}

export function writeState(dataPath, state) {
  const file = resolveDataPath(dataPath);
  mkdirSync(dirname(file), { recursive: true });
  writeFileSync(file, JSON.stringify(state, null, 2) + '\n', 'utf8');
}

export function update(dataPath, mutator) {
  const state = readState(dataPath);
  const result = mutator(state);
  writeState(dataPath, state);
  return result;
}

import { LibraryError } from './errors.js';
import { addDays, diffDays, isValidDate, todayISO } from './dates.js';
import { readState, update } from './store.js';

const MAX_ACTIVE_LOANS = 3;
const FINE_PER_DAY = 0.5;
const MAX_FINE = 20;

function requireDate(value, fallback) {
  const date = value === undefined ? fallback : value;
  if (!isValidDate(date)) throw new LibraryError(`invalid date: ${value}`);
  return date;
}

function requireId(value, field) {
  if (!Number.isInteger(value) || value < 1) throw new LibraryError(`${field} must be a positive integer`);
  return value;
}

function activeLoansFor(state, memberId) {
  return state.loans.filter((l) => l.memberId === memberId && l.returned === null);
}

function fineFor(due, returned) {
  const late = diffDays(due, returned);
  if (late <= 0) return 0;
  return Math.round(Math.min(late * FINE_PER_DAY, MAX_FINE) * 100) / 100;
}

export function checkout({ bookId, memberId, days = 14, today, data } = {}) {
  const out = requireDate(today, todayISO());
  requireId(bookId, 'bookId');
  requireId(memberId, 'memberId');
  if (!Number.isInteger(days) || days < 1) throw new LibraryError('days must be a positive integer');
  return update(data, (state) => {
    const book = state.books.find((b) => b.id === bookId);
    if (!book) throw new LibraryError(`book ${bookId} not found`);
    const member = state.members.find((m) => m.id === memberId);
    if (!member) throw new LibraryError(`member ${memberId} not found`);
    if (book.available < 1) throw new LibraryError(`book ${bookId} has no available copies`);
    if (activeLoansFor(state, memberId).length >= MAX_ACTIVE_LOANS) {
      throw new LibraryError(`member ${memberId} already has ${MAX_ACTIVE_LOANS} active loans`);
    }
    book.available -= 1;
    const id = state.sequences.loans + 1;
    state.sequences.loans = id;
    const loan = { id, bookId, memberId, out, due: addDays(out, days), returned: null, fine: 0 };
    state.loans.push(loan);
    return loan;
  });
}

export function returnLoan({ loanId, today, data } = {}) {
  requireId(loanId, 'loanId');
  const returned = requireDate(today, todayISO());
  return update(data, (state) => {
    const loan = state.loans.find((l) => l.id === loanId);
    if (!loan) throw new LibraryError(`loan ${loanId} not found`);
    if (loan.returned !== null) throw new LibraryError(`loan ${loanId} has already been returned`);
    loan.returned = returned;
    loan.fine = fineFor(loan.due, returned);
    const book = state.books.find((b) => b.id === loan.bookId);
    if (book) book.available = Math.min(book.copies, book.available + 1);
    return loan;
  });
}

export function listLoans({ memberId, overdue, today, data } = {}) {
  const state = readState(data);
  let loans = state.loans.slice();
  if (memberId !== undefined) {
    requireId(memberId, 'memberId');
    loans = loans.filter((l) => l.memberId === memberId);
  }
  if (overdue) {
    const now = requireDate(today, todayISO());
    loans = loans.filter((l) => l.returned === null && l.due < now);
  }
  return loans;
}

export function report({ today, data } = {}) {
  const state = readState(data);
  const now = requireDate(today, todayISO());
  let copies = 0, available = 0;
  for (const b of state.books) { copies += b.copies; available += b.available; }
  const active = state.loans.filter((l) => l.returned === null);
  const overdue = active.filter((l) => l.due < now).length;
  let finesOwed = 0;
  for (const l of state.loans) finesOwed += l.fine || 0;
  finesOwed = Math.round(finesOwed * 100) / 100;
  return { books: state.books.length, copies, available, members: state.members.length, activeLoans: active.length, overdue, finesOwed };
}

function parseOptions(argv, spec) {
  const opts = {};
  let i = 0;
  while (i < argv.length) {
    const token = argv[i];
    i += 1;
    if (!token.startsWith('--')) {
      throw new UsageError(`unexpected argument: ${token}`);
    }
    let name = token.slice(2);
    let inline;
    const eq = name.indexOf('=');
    if (eq !== -1) { inline = name.slice(eq + 1); name = name.slice(0, eq); }
    if (name === 'data') {
      opts.data = takeValue(argv, i, inline, 'data');
      if (inline === undefined) i += 1;
      continue;
    }
    const def = spec.options.find((o) => o.flag === name);
    if (!def) throw new UsageError(`unknown option --${name}`);
    if (def.type === 'boolean') {
      opts[def.key] = inline === undefined ? true : toBool(inline, name);
      continue;
    }
    const value = takeValue(argv, i, inline, name);
    if (inline === undefined) i += 1;
    opts[def.key] = def.type === 'int' ? toPositiveInt(value, name) : value;
  }
  for (const def of spec.options) {
    if (def.required && opts[def.key] === undefined) {
      throw new UsageError(`missing required option --${def.flag}`);
    }
  }
  return opts;
}

function takeValue(argv, index, inline, name) {
  if (inline !== undefined) return inline;
  const value = argv[index];
  if (value === undefined || value.startsWith('--')) {
    throw new UsageError(`option --${name} requires a value`);
  }
  return value;
}
