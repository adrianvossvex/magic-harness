function parseOptions(args, spec) {
  const opts = {};
  for (let i=0;i<args.length;i++){
    const a = args[i];
    if (!a.startsWith('--')) throw new UsageError(`Unexpected argument: ${a}`);
    let key, val;
    const eq = a.indexOf('=');
    if (eq !== -1) { key = a.slice(2, eq); val = a.slice(eq+1); }
    else { key = a.slice(2); }
    if (spec.flags.includes(key)) {
      if (val !== undefined) throw new UsageError(`Option --${key} does not take a value`);
      opts[key] = true;
      continue;
    }
    if (!spec.options.includes(key)) throw new UsageError(`Unknown option: --${key}`);
    if (val === undefined) {
      if (i+1 >= args.length || args[i+1].startsWith('--')) throw new UsageError(`Missing value for --${key}`);
      val = args[++i];
    }
    opts[key] = val;
  }
  return opts;
}

  const y = date.getFullYear();
  const m = String(date.getMonth()+1).padStart(2,'0');
  const d = String(date.getDate()).padStart(2,'0');
  return `${y}-${m}-${d}`;

export function parseDate(s, name='date') {

export function nextId(db, type){
  const seq = db.sequences;
  if (typeof seq[type] !== 'number') {
    seq[type] = db[type].reduce((m,x)=>Math.max(m,x.id||0),0);

  seq[type] += 1;

export function checkout({ bookId, memberId, days = 14, today } = {}) {
  const out = today ?? todayString();
  const n = Number(days);
  if (!Number.isInteger(n) || n < 1) throw new AppError('days must be a positive integer');
  return update(db => {
    const book = db.books.find(b => b.id === bookId);
    if (!book) throw new AppError(`Book not found: ${bookId}`);
    const member = db.members.find(m => m.id === memberId);
    if (!member) throw new AppError(`Member not found: ${memberId}`);

    db.loans.push(loan);
    return { ...loan };

export function returnLoan({ loanId, today } = {}) {
  const ret = today ?? todayString();
  return update(db => {
    const loan = db.loans.find(l => l.id === loanId);
    if (!loan) throw new AppError(`Loan not found: ${loanId}`);
    if (loan.returned !== null) throw new AppError(`Loan ${loanId} has already been returned`);
    const book = db.books.find(b => b.id === loan.bookId);
    const late = daysBetween(loan.due, ret);
    let fine = 0;
    if (late > 0) fine = Math.min(late * FINE_PER_DAY, FINE_CAP);
    loan.returned = ret;
    loan.fine = fine;
    if (book) book.available = Math.min(book.available + 1, book.copies);
    return { ...loan };
  });
}

function parse(args, spec) {
  const opts = {};
  const values = new Set([...(spec.values||[]), 'data']);
  const flags = new Set(spec.flags||[]);
  for (let i=0;i<args.length;i++){
    const a = args[i];
    if (typeof a !== 'string' || !a.startsWith('--') || a === '--') throw new UsageError(`Unexpected argument: ${a}`);

    if (eq >= 0) { key = a.slice(2, eq); val = a.slice(eq+1); }
    else key = a.slice(2);
    if (flags.has(key)) {
      if (val !== undefined) throw new UsageError(`Option --${key} does not take a value`);
      opts[key] = true;

  }
  for (const req of spec.required || []) {
    if (opts[req] === undefined) throw new UsageError(`Missing required option: --${req}`);

  return opts;
}
